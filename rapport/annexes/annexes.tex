\chapter{Extrait de la table de règles}
\label{ann:regles}

La règle \cmd{gzip}, ajoutée par M2.3a après la découverte de V20. Chaque \cmd{ex(\ldots)} est un
test : la commande est simulée dans l'arborescence de référence, et l'impact doit être
exactement celui indiqué. Les mentions \cmd{VERIFIED} décrivent un comportement observé sur des
fichiers jetables.

\begin{code}{\fichier{src/demo\_cli/engine/rules.py} (extrait)}
Rule(id="gzip", tools=("gzip", "gunzip", "zcat"), resolver="compress",
     options=OptionSpec(flags=_GZ_FLAGS, values={"-S": "suffix", "--suffix": "suffix"}),
     examples=(
         ex("gzip a.txt", destroys={"a.txt"}, creates={"a.txt.gz"},
            note="VERIFIED: the input is removed once a.txt.gz is written"),
         ex("gzip -k a.txt", creates={"a.txt.gz"}),
         ex("gzip -c a.txt > /dev/null", note="to stdout"),
         ex("touch a.txt.gz; gzip a.txt", creates={"a.txt.gz"},
            note="VERIFIED: an existing output is not overwritten (no terminal)"),
         ex("touch a.txt.gz; gzip -f a.txt", destroys={"a.txt"}, creates={"a.txt.gz"}),
         ex("gzip -r dir", destroys={"dir/"}, note="VERIFIED: -r rewrites every file under it"),
         ex("gzip --rm a.txt", unknown=True, note="an option demo_cli does not know"),
         ex("gzip $(cat list.txt)", destroys={"a.txt"}, creates={"a.txt.gz"}),
     )),
\end{code}

\chapter{Extraits de tests}
\label{ann:tests}

\section*{V17 : pas de point de contrôle pour un fichier hors du projet}
\begin{code}{\fichier{tests/test\_v17.py} (en-tête)}
"""V17 [...]: with checkpoints enabled, a line the engine cannot read whose
operand lies OUTSIDE the project took a whole-project checkpoint and came back
REVERSIBLE - but the checkpoint holds only the project, so the outside file was
not recoverable. Such a line now escalates; the same line inside the project
still gets its checkpoint."""
\end{code}

\section*{M2.3a : une ligne non lue va au filet, sinon au point de contrôle, sinon escalade}
\begin{code}{\fichier{tests/test\_m23a.py}}
def test_6_an_unread_line_takes_the_net(tmp_path, net):
    proj = _project(tmp_path)
    (proj / "cmd.txt").write_text("rm -rf build\n")
    r = _eval(proj, 'sh -c "$(cat cmd.txt)"', net=True)
    assert (r.decision.decision, r.classification.matched_rule) == ("SUPERVISED", "not_understood")

def test_6a_with_no_net_a_checkpoint_else_it_escalates(tmp_path):
    proj = _project(tmp_path)
    (proj / "cmd.txt").write_text("rm -rf build\n")
    line = 'sh -c "$(cat cmd.txt)"'
    r = _eval(proj, line, checkpoints=True)
    assert (r.decision.decision, r.classification.matched_rule) == ("REVERSIBLE", "not_understood")
    assert r.recovery_entry is not None
    r = _eval(proj, line)
    assert (r.decision.decision, r.recovery_entry) == ("ESCALATE", None)
\end{code}
